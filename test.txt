# Options for analysis running.
version: "2"
run:
  # The default timeout is 1m. Set a longer timeout for larger projects.
  timeout: 5m
  # By default, it will analyze all Go files in the current directory and subdirectories.
  # You can specify directories and files to analyze here.
  # paths:
  #   - 'cmd/'
  #   - 'pkg/'

# Linter-specific settings and configuration.
linters:
  # Enable specific linters.
  # To see a full list of available linters and their descriptions:
  # $ golangci-lint help linters
  enable:
    - gofmt      # Checks if code is formatted with gofmt.
    - govet      # Reports suspicious constructs.
    - errcheck   # Checks for unchecked errors.
    - staticcheck # A huge suite of powerful static analysis checks.
    - unused     # Checks for unused code.
    - ineffassign # Detects ineffectual assignments.
    - gosimple   # Recommends simplifications to your code.
    - stylecheck # A collection of style checks.

# Configuration for specific issues or rules.
issues:
  # Maximum issues of the same type to report.
  max-same-issues: 10

  # Exclude specific rules for specific paths. This is useful for ignoring
  # certain linter warnings in generated code or test files.
  exclude-rules:
    - path: _test\.go
      linters:
        - errcheck # Don't require error checking in test files.
        - funlen   # Test functions can be long.

# Output configuration.
output:
  # Print colored output.
  print-colored-line: true
